        const container = document.getElementById('canvas-container');

        const camera = new THREE.PerspectiveCamera(75, window.innerWidth / window.innerHeight, 0.1, 1000);

        container.appendChild(renderer.domElement);

        // Lighting Architecture

        scene.add(ambientLight);

        const flashlight = new THREE.SpotLight(0xffffff, 1.5, 20, Math.PI / 6, 0.5, 1);
        camera.add(flashlight);
        flashlight.position.set(0, 0, 0);

        camera.add(flashlight.target);
        scene.add(camera);

        // World Hierarchy Groups

        /* =========================================================================
           3. MAP ARCHITECTURE BUILDER (HOSPITAL LAYOUT)
           ========================================================================= */
        function createWall(x, z, width, depth, height = 4, color = 0xdcdde1) {
            const mat = new THREE.MeshStandardMaterial({ color: color, roughness: 0.4 });
            const geo = new THREE.BoxGeometry(width, height, depth);
            const wall = new THREE.Mesh(geo, mat);
            wall.position.set(x, height / 2, z);
            wall.castShadow = true;
            wall.receiveShadow = true;
            worldGroup.add(wall);
            return wall;
        }

        function buildHospitalMap() {
            // Main Floor
            const floorMat = new THREE.MeshStandardMaterial({ color: 0x333344, roughness: 0.2 });
            const floorGeo = new THREE.PlaneGeometry(60, 60);
            const floor = new THREE.Mesh(floorGeo, floorMat);
            floor.rotation.x = -Math.PI / 2;
            floor.receiveShadow = true;
            worldGroup.add(floor);

            // Ceiling
            const ceilingMat = new THREE.MeshStandardMaterial({ color: 0x222222, roughness: 0.8 });
            const ceilingGeo = new THREE.PlaneGeometry(60, 60);
            const ceiling = new THREE.Mesh(ceilingGeo, ceilingMat);
            ceiling.rotation.x = Math.PI / 2;
            ceiling.position.y = 4;
            worldGroup.add(ceiling);

            // Perimeter Walls
            createWall(0, -30, 60, 1);
            createWall(0, 30, 60, 1);
            createWall(-30, 0, 1, 60);
            createWall(30, 0, 1, 60);

            // Interior Partitions (Rooms 101, 102, 103 & Emergency & Office)
            createWall(-10, -10, 1, 40); // Main Corridor Left
            createWall(10, -10, 1, 40);  // Main Corridor Right

            // Ward Beds Rooms (Left Wing)
            for(let i = 0; i < 3; i++) {
                let zPos = -20 + (i * 12);
                createWall(-20, zPos, 20, 0.8);
                createBed(-22, zPos + 4, `Room ${101 + i}`);
            }

            // Registration Counter Glass Box & Desk
            const deskMat = new THREE.MeshStandardMaterial({ color: 0x5c3a21 });
            const desk = new THREE.Mesh(new THREE.BoxGeometry(8, 1.2, 2), deskMat);
            desk.position.set(0, 0.6, 5);
            worldGroup.add(desk);

            // Registration Computer Monitor
            const compMat = new THREE.MeshStandardMaterial({ color: 0x111111 });
            const comp = new THREE.Mesh(new THREE.BoxGeometry(1.2, 1, 0.8), compMat);
            comp.position.set(-1.5, 1.7, 5);
            comp.name = "Computer Terminal";
            comp.userData = { interactable: true, type: "computer" };
            interactablesGroup.add(comp);

            // Tag Printer Device
            const printMat = new THREE.MeshStandardMaterial({ color: 0xdddddd });
            const printer = new THREE.Mesh(new THREE.BoxGeometry(1, 0.6, 0.8), printMat);
            printer.position.set(2, 1.5, 5);
            printer.name = "ID Tag Printer";
            printer.userData = { interactable: true, type: "printer" };
            interactablesGroup.add(printer);

            // Milk Dispenser Station
            const milkMat = new THREE.MeshStandardMaterial({ color: 0xffffff, emissive: 0x112233 });
            const milkStation = new THREE.Mesh(new THREE.BoxGeometry(1.2, 2, 1.2), milkMat);
            milkStation.position.set(6, 1, 6);
            milkStation.name = "Milk Station";
            milkStation.userData = { interactable: true, type: "milk" };
            interactablesGroup.add(milkStation);

            // Supplies Shelf (10 Medicines)
            buildSupplyShelves();

            // Point Lights along Ceiling Corridor
            for(let z = -20; z <= 20; z += 10) {
                const light = new THREE.PointLight(0xffeedd, 0.8, 15);
                light.position.set(0, 3.8, z);
                worldGroup.add(light);
            }
        }

        function createBed(x, z, roomName) {
            const bedFrame = new THREE.Mesh(
                new THREE.BoxGeometry(3, 0.8, 5),
                new THREE.MeshStandardMaterial({ color: 0x778899 })
            );
            bedFrame.position.set(x, 0.4, z);

            const mattress = new THREE.Mesh(
                new THREE.BoxGeometry(2.8, 0.4, 4.8),
                new THREE.MeshStandardMaterial({ color: 0xffffff })
            );
            mattress.position.set(x, 0.8, z);
            worldGroup.add(bedFrame);
            worldGroup.add(mattress);

            // Bed Node Marker for interaction
            const bedNode = new THREE.Mesh(
                new THREE.BoxGeometry(1, 1, 1),
                new THREE.MeshBasicMaterial({ visible: false })
            );
            bedNode.position.set(x, 1, z);
            bedNode.name = `${roomName} Bed`;
            bedNode.userData = { interactable: true, type: "bed", room: roomName, occupied: false };
            interactablesGroup.add(bedNode);
        }

        function buildSupplyShelves() {
            const shelfMat = new THREE.MeshStandardMaterial({ color: 0x444444 });
            const shelf = new THREE.Mesh(new THREE.BoxGeometry(1, 3, 12), shelfMat);
            shelf.position.set(9, 1.5, -5);
            worldGroup.add(shelf);

            CONFIG.medicines.forEach((medName, idx) => {
                const itemGeo = new THREE.CylinderGeometry(0.2, 0.2, 0.5);
                const itemMat = new THREE.MeshStandardMaterial({ color: Math.random() * 0xffffff });
                const itemMesh = new THREE.Mesh(itemGeo, itemMat);
                itemMesh.position.set(8.3, 0.6 + ((idx % 4) * 0.7), -9 + (Math.floor(idx / 4) * 3));
                itemMesh.name = medName;
                itemMesh.userData = { interactable: true, type: "medicine", medName: medName };
                interactablesGroup.add(itemMesh);
            });
        }

        /* =========================================================================
           4. PROCEDURAL 3D ANIMAL PATIENT MESH GENERATOR
           ========================================================================= */
        function createPatientMesh(speciesType, colorHex, isAnomaly, anomalyType) {
            const group = new THREE.Group();

            // Base Body
            const bodyMat = new THREE.MeshStandardMaterial({ color: colorHex, roughness: 0.5 });
            const body = new THREE.Mesh(new THREE.CapsuleGeometry(0.5, 1, 4, 8), bodyMat);
            body.position.y = 1;
            group.add(body);

            // Head
            const head = new THREE.Mesh(new THREE.SphereGeometry(0.45, 16, 16), bodyMat);
            head.position.set(0, 1.8, 0);
            group.add(head);

            // Ears Procedural Morphing
            const earMat = bodyMat;
            if (speciesType === "Rabbit") {
                const earGeo = new THREE.CylinderGeometry(0.08, 0.08, 0.8);
                const earL = new THREE.Mesh(earGeo, earMat);
                earL.position.set(-0.2, 2.4, 0);
                const earR = earL.clone();
                earR.position.set(0.2, 2.4, 0);
                group.add(earL, earR);
            } else if (speciesType === "Cat" || speciesType === "Devil-Cat") {
                const earGeo = new THREE.ConeGeometry(0.15, 0.4, 4);
                const earL = new THREE.Mesh(earGeo, earMat);
                earL.position.set(-0.25, 2.3, 0);
                const earR = earL.clone();
                earR.position.set(0.25, 2.3, 0);
                group.add(earL, earR);
                if (speciesType === "Devil-Cat") {
                    const hornMat = new THREE.MeshStandardMaterial({ color: 0xaa2222 });
                    const hornL = new THREE.Mesh(new THREE.ConeGeometry(0.08, 0.3, 4), hornMat);
                    hornL.position.set(-0.1, 2.3, 0.1);
                    group.add(hornL);
                }
            } else if (speciesType === "Bear" || speciesType === "Dog") {
                const earGeo = new THREE.SphereGeometry(0.15, 8, 8);
                const earL = new THREE.Mesh(earGeo, earMat);
                earL.position.set(-0.35, 2.2, 0);
                const earR = earL.clone();
                earR.position.set(0.35, 2.2, 0);
                group.add(earL, earR);
            }

            // Uniform Deep Slate Almond Eyes
            const eyeColor = (isAnomaly && anomalyType === "instant") ? 0xff0000 : 0x111116;
            const eyeMat = new THREE.MeshBasicMaterial({ color: eyeColor });
            const eyeGeo = new THREE.SphereGeometry(0.08, 8, 8);
            
            const eyeL = new THREE.Mesh(eyeGeo, eyeMat);
            eyeL.position.set(-0.18, 1.85, 0.38);
            const eyeR = eyeL.clone();
            eyeR.position.set(0.18, 1.85, 0.38);
            group.add(eyeL, eyeR);

            group.userData = {
                species: speciesType,
                color: colorHex,
                isAnomaly: isAnomaly,
                anomalyType: anomalyType,
                eyeMeshL: eyeL,
                eyeMeshR: eyeR
            };

            return group;
        }

        /* =========================================================================
           5. FIRST-PERSON CONTROLLER & INPUT HANDLING
           ========================================================================= */
        const moveState = { forward: false, backward: false, left: false, right: false, run: false };
        let cameraRotation = { yaw: 0, pitch: 0 };
        const playerSpeed = 0.1;
        const playerPos = new THREE.Vector3(0, 1.6, 20);

        // Mouse Look & Pointer Lock
        document.addEventListener('mousemove', (e) => {
            if (document.pointerLockElement === document.body) {
                cameraRotation.yaw -= e.movementX * 0.002;
                cameraRotation.pitch -= e.movementY * 0.002;
                cameraRotation.pitch = Math.max(-Math.PI / 2.5, Math.min(Math.PI / 2.5, cameraRotation.pitch));
                
                // Don't look up monster penalty check
                if (cameraRotation.pitch > 0.6 && state.ceilingMonsterActive) {
                    state.sanity = Math.max(0, state.sanity - 1.5);
                    updateHUD();
                }
            }
        });

        document.addEventListener('keydown', (e) => {
            if (!state.inGame) return;
            switch(e.code) {
                case 'KeyW': moveState.forward = true; break;
                case 'KeyS': moveState.backward = true; break;
                case 'KeyA': moveState.left = true; break;
                case 'KeyD': moveState.right = true; break;
                case 'ShiftLeft': moveState.run = true; break;
                case 'KeyE': triggerInteraction(); break;
                case 'KeyF': toggleFlashlight(); break;
                case 'Digit1': selectSlot(0); break;
                case 'Digit2': selectSlot(1); break;
                case 'Digit3': selectSlot(2); break;
            }
        });

        document.addEventListener('keyup', (e) => {
            switch(e.code) {
                case 'KeyW': moveState.forward = false; break;
                case 'KeyS': moveState.backward = false; break;
                case 'KeyA': moveState.left = false; break;
                case 'KeyD': moveState.right = false; break;
                case 'ShiftLeft': moveState.run = false; break;
            }
        });

        function updatePlayerMovement() {
            if (!state.inGame) return;

            const speed = playerSpeed * (moveState.run ? 1.5 : 1.0);
            const euler = new THREE.Euler(0, cameraRotation.yaw, 0, 'YXZ');
            const dir = new THREE.Vector3();

            if (moveState.forward) dir.z -= 1;
            if (moveState.backward) dir.z += 1;
            if (moveState.left) dir.x -= 1;
            if (moveState.right) dir.x += 1;

            dir.normalize();
            dir.applyEuler(euler);
            playerPos.addScaledVector(dir, speed);

            // Simple Axis-Aligned Bounding Box Collisions for Hospital Boundaries
            playerPos.x = Math.max(-28, Math.min(28, playerPos.x));
            playerPos.z = Math.max(-28, Math.min(28, playerPos.z));

            camera.position.copy(playerPos);
            camera.rotation.set(cameraRotation.pitch, cameraRotation.yaw, 0, 'YXZ');

            checkRaycastInteractions();
        }

        /* =========================================================================
           6. RAYCASTING & OBJECT INTERACTION SYSTEM
           ========================================================================= */
        const raycaster = new THREE.Raycaster();
        let hoveredObject = null;

        function checkRaycastInteractions() {
            raycaster.setFromCamera(new THREE.Vector2(0, 0), camera);
            const intersects = raycaster.intersectObjects(interactablesGroup.children, true);

            const promptEl = document.getElementById('interaction-prompt');

            if (intersects.length > 0 && intersects[0].distance < 4) {
                let obj = intersects[0].object;
                while (obj.parent && obj.parent !== interactablesGroup && !obj.userData.interactable) {
                    obj = obj.parent;
                }
                if (obj.userData && obj.userData.interactable) {
                    hoveredObject = obj;
                    promptEl.style.display = 'block';
                    promptEl.innerText = `Press [E] to interact with ${obj.name}`;
                    return;
                }
            }
            hoveredObject = null;
            promptEl.style.display = 'none';
        }

        function triggerInteraction() {
            if (!hoveredObject) return;

            const data = hoveredObject.userData;
            if (data.type === "computer") {
                openComputerUI();
            } else if (data.type === "printer") {
                printPatientTag();
            } else if (data.type === "milk") {
                drinkMilk();
            } else if (data.type === "medicine") {
                pickupItem(data.medName);
            } else if (data.type === "bed") {
                interactWithBed(hoveredObject);
            }
        }

        /* =========================================================================
           7. GAMEPLAY LOGIC & SHIFT MANAGEMENT
           ========================================================================= */
        function startNewShift(shiftNum) {
            state.shift = shiftNum;
            state.sanity = 100;
            state.strikes = 3;
            state.isShiftActive = true;
            state.inGame = true;
            state.ceilingMonsterActive = (shiftNum >= 2);

            document.getElementById('main-menu').style.display = 'none';
            document.getElementById('summary-menu').style.display = 'none';
            document.body.requestPointerLock();

            updateHUD();
            spawnNextPatientInQueue();
        }

        function spawnNextPatientInQueue() {
            if (patientsGroup.children.length > 0) return;

            const speciesList = ["Rabbit", "Cat", "Dog", "Bear", "Devil-Cat"];
            const species = speciesList[Math.floor(Math.random() * speciesList.length)];
            const color = CONFIG.palette[Math.floor(Math.random() * CONFIG.palette.length)];
            
            // Anomaly Probabilities
            const rand = Math.random();
            let isAnomaly = false;
            let anomalyType = "none";

            if (state.shift > 1) {
                if (rand < 0.15) { isAnomaly = true; anomalyType = "cctv"; }
                else if (rand < 0.25) { isAnomaly = true; anomalyType = "photo"; }
                else if (rand < 0.30) { isAnomaly = true; anomalyType = "uv"; }
            }

            const patientMesh = createPatientMesh(species, color, isAnomaly, anomalyType);
            patientMesh.position.set(0, 0, 8); // At glass window counter
            patientMesh.rotation.y = Math.PI; // Face player directly
            patientsGroup.add(patientMesh);

            state.currentPatient = {
                mesh: patientMesh,
                name: `Patient #${Math.floor(Math.random() * 899 + 100)}`,
                requiredMedicines: [
                    CONFIG.medicines[Math.floor(Math.random() * CONFIG.medicines.length)]
                ],
                tagPrinted: false,
                isAnomaly: isAnomaly,
                anomalyType: anomalyType
            };

            setObjective(`Inspect ${state.currentPatient.name} at Registration Desk`);
        }

        function openComputerUI() {
            document.exitPointerLock();
            document.getElementById('pc-interface').style.display = 'flex';
            
            // Render eyes in CCTV preview according to anomaly logic
            const camEyes = document.getElementById('cam-eyes');
            if (state.currentPatient && state.currentPatient.isAnomaly && state.currentPatient.anomalyType === "cctv") {
                camEyes.children[0].style.background = '#ff0000';
                camEyes.children[1].style.background = '#ff0000';
            } else {
                camEyes.children[0].style.background = '#000000';
                camEyes.children[1].style.background = '#000000';
            }
        }

        function printPatientTag() {
            if (!state.currentPatient) return;
            state.currentPatient.tagPrinted = true;
            pickupItem(`ID Tag: ${state.currentPatient.name}`);
            setObjective(`Assign ${state.currentPatient.name} to Ward Room Bed`);
        }

        function pickupItem(itemName) {
            for (let i = 0; i < 3; i++) {
                if (!state.inventory[i]) {
                    state.inventory[i] = itemName;
                    updateInventoryUI();
                    return;
                }
            }
            alert("Inventory Full!");
        }

        function selectSlot(index) {
            state.activeSlot = index;
            updateInventoryUI();
        }

        function updateInventoryUI() {
            for (let i = 0; i < 3; i++) {
                const slot = document.getElementById(`slot-${i}`);
                slot.innerText = state.inventory[i] ? state.inventory[i] : "Empty";
                slot.classList.toggle('active', i === state.activeSlot);
            }
        }

        function drinkMilk() {
            if (state.milkCooldown > 0) return;
            state.sanity = Math.min(100, state.sanity + 25);
            state.milkCooldown = 35;
            updateHUD();
        }

        function interactWithBed(bedMesh) {
            const heldItem = state.inventory[state.activeSlot];
            
            // Assign Patient to Bed
            if (heldItem && heldItem.startsWith("ID Tag") && state.currentPatient) {
                state.inventory[state.activeSlot] = null;
                updateInventoryUI();
                
                // Move patient mesh to bed
                state.currentPatient.mesh.position.set(bedMesh.position.x, 0.8, bedMesh.position.z);
                state.currentPatient.mesh.rotation.x = -Math.PI / 2; // Lie down
                bedMesh.userData.occupied = true;
                bedMesh.userData.patientData = state.currentPatient;

                setObjective(`Administer ${state.currentPatient.requiredMedicines.join(", ")} to ${state.currentPatient.name}`);
                return;
            }

            // Deliver Medicine Bedside
            if (bedMesh.userData.occupied && heldItem) {
                const patient = bedMesh.userData.patientData;
                const reqIdx = patient.requiredMedicines.indexOf(heldItem);

                if (reqIdx !== -1) {
                    // Correct Medicine
                    patient.requiredMedicines.splice(reqIdx, 1);
                    state.inventory[state.activeSlot] = null;
                    updateInventoryUI();

                    if (patient.requiredMedicines.length === 0) {
                        handlePatientRecovery(patient, bedMesh);
                    }
                } else {
                    // Incorrect Medicine -> Kills Patient
                    handleSafetyStrike("Administered wrong medicine!");
                    patientsGroup.remove(patient.mesh);
                    bedMesh.userData.occupied = false;
                    setTimeout(spawnNextPatientInQueue, 2000);
                }
            }
        }

        function handlePatientRecovery(patient, bedMesh) {
            if (patient.isAnomaly) {
                // Anomaly Awakening Event
                patient.mesh.userData.eyeMeshL.material.color.setHex(0xff0000);
                patient.mesh.userData.eyeMeshR.material.color.setHex(0xff0000);
                
                setTimeout(() => {
                    // Attack player sanity
                    state.sanity = Math.max(0, state.sanity - 35);
                    updateHUD();
                    patientsGroup.remove(patient.mesh);
                    bedMesh.userData.occupied = false;
                    setTimeout(spawnNextPatientInQueue, 2000);
                }, 2000);
            } else {
                // Normal Patient Leaves Cleanly
                state.stats.recovered++;
                setTimeout(() => {
                    patientsGroup.remove(patient.mesh);
                    bedMesh.userData.occupied = false;
                    setTimeout(spawnNextPatientInQueue, 2000);
                }, 1000);
            }
        }

        function handleSafetyStrike(reason) {
            state.strikes--;
            state.stats.deaths++;
            updateHUD();

            if (state.strikes <= 0) {
                state.isFired = true;
                state.inGame = false;
                alert("GAME OVER: You were fired for safety violations!");
                location.reload();
            }
        }

        function toggleFlashlight() {
            flashlight.visible = !flashlight.visible;
        }

        function setObjective(text) {
            state.currentObjective = text;
            document.getElementById('objective-text').innerText = text;
        }

        function updateHUD() {
            document.getElementById('shift-display').innerText = `SHIFT ${state.shift}`;
            document.getElementById('sanity-bar').style.width = `${state.sanity}%`;
            
            let hearts = "";
            for (let i = 0; i < state.strikes; i++) hearts += "❤️";
            document.getElementById('strikes-display').innerText = `SAFETY STRIKES: ${hearts}`;

            const warningBanner = document.getElementById('warning-banner');
            warningBanner.style.display = (state.ceilingMonsterActive && cameraRotation.pitch > 0.4) ? 'block' : 'none';
        }

           ========================================================================= */
        document.getElementById('btn-start').addEventListener('click', () => startNewShift(1));
        document.getElementById('btn-close-pc').addEventListener('click', () => {
            document.getElementById('pc-interface').style.display = 'none';
            document.body.requestPointerLock();

               (state.currentPatient.anomalyType === "photo" || state.currentPatient.anomalyType === "cctv")) {
                polEyes.forEach(e => e.style.background = '#ff0000');
            } else {
                polEyes.forEach(e => e.style.background = '#000000');
            }

        });

            document.getElementById('polaroid-view').style.display = 'none';
        });

        document.getElementById('btn-decline-patient').addEventListener('click', () => {
            if (state.currentPatient) {
                state.stats.declined++;
                patientsGroup.remove(state.currentPatient.mesh);
                state.currentPatient = null;
                document.getElementById('pc-interface').style.display = 'none';
                document.body.requestPointerLock();
                setTimeout(spawnNextPatientInQueue, 2000);
            }
        });

        /* =========================================================================
           9. MAIN GAME LOOP
           ========================================================================= */
        let lastTime = performance.now();

        function animate() {
            requestAnimationFrame(animate);

            const now = performance.now();
            const delta = (now - lastTime) / 1000;
            lastTime = now;

            if (state.inGame) {
                updatePlayerMovement();

                // Timers
                if (state.milkCooldown > 0) state.milkCooldown -= delta;
                
                // Slowly decay sanity over time
                state.sanity = Math.max(0, state.sanity - (delta * 0.5));
                updateHUD();

                if (state.sanity <= 0 && !state.isFired) {
                    state.isFired = true;
                    alert("GAME OVER: You succumbed to insanity!");
                    location.reload();
                }
            }

            renderer.render(scene, camera);
        }

        // Initialize Map & Start Engine
        buildHospitalMap();
        animate();

        // Responsive Resize Handler
        window.addEventListener('resize', () => {
            camera.aspect = window.innerWidth / window.innerHeight;
            camera.updateProjectionMatrix();
            renderer.setSize(window.innerWidth, window.innerHeight);
        });
